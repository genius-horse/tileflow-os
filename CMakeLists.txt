cmake_minimum_required(VERSION 3.16)
project(TileFlowOS LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "TileFlow OS requires Linux (native, WSL2, or a Linux server).")
endif()
find_package(Threads REQUIRED)
find_package(OpenCV 4.2 REQUIRED COMPONENTS core imgcodecs imgproc)

add_library(tileflow_core src/disk.cpp src/worker.cpp)
target_include_directories(tileflow_core PUBLIC include)
target_include_directories(tileflow_core SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(tileflow_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
target_compile_features(tileflow_core PUBLIC cxx_std_17)
target_compile_options(tileflow_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(tileflow src/main.cpp src/service.cpp)
target_include_directories(tileflow SYSTEM PRIVATE third_party)
target_link_libraries(tileflow PRIVATE tileflow_core)
target_compile_options(tileflow PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(tileflow PRIVATE TILEFLOW_WEB_DIR="${CMAKE_CURRENT_SOURCE_DIR}/web")

include(CTest)
foreach(name IN ITEMS disk image)
  if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}_test.cpp")
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE tileflow_core)
    add_test(NAME ${name} COMMAND ${name}_test)
  endif()
endforeach()
